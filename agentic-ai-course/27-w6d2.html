Should you build an MCP server? (lecture 7)

"The number one advantage is you make an MCP server if you want to share it": you write the descriptions once and anyone can plug it in, together with resources ("a bit like a RAG context") and prompt templates. But "if you're only building a tool for you to use yourself… there's no point in building an MCP server, that's wasting time. You can simply decorate that tool with @function_tool." A server is a separate process talking over stdio: extra machinery for no gain.

from pydantic import BaseModel
import json
from dotenv import load_dotenv
from datetime import datetime
from market import get_share_price
from database import write_account, read_account, write_log

load_dotenv(override=True)

INITIAL_BALANCE = 10_000.0
SPREAD = 0.002


class Transaction(BaseModel):
    symbol: str
    quantity: int
    price: float
    timestamp: str
    rationale: str

    def total(self) -> float:
        return self.quantity * self.price
    
    def __repr__(self):
        return f"{abs(self.quantity)} shares of {self.symbol} at {self.price} each."


class Account(BaseModel):
    name: str
    balance: float
    strategy: str
    holdings: dict[str, int]
    transactions: list[Transaction]
    portfolio_value_time_series: list[tuple[str, float]]

    @classmethod
    def get(cls, name: str):
        fields = read_account(name.lower())
        if not fields:
            fields = {
                "name": name.lower(),
                "balance": INITIAL_BALANCE,
                "strategy": "",
                "holdings": {},
                "transactions": [],
                "portfolio_value_time_series": []
            }
            write_account(name, fields)
        return cls(**fields)
    
    
    def save(self):
        write_account(self.name.lower(), self.model_dump())

    def reset(self, strategy: str):
        self.balance = INITIAL_BALANCE
        self.strategy = strategy
        self.holdings = {}
        self.transactions = []
        self.portfolio_value_time_series = []
        self.save()

    def deposit(self, amount: float):
        """ Deposit funds into the account. """
        if amount <= 0:
            raise ValueError("Deposit amount must be positive.")
        self.balance += amount
        print(f"Deposited ${amount}. New balance: ${self.balance}")
        self.save()

    def withdraw(self, amount: float):
        """ Withdraw funds from the account, ensuring it doesn't go negative. """
        if amount > self.balance:
            raise ValueError("Insufficient funds for withdrawal.")
        self.balance -= amount
        print(f"Withdrew ${amount}. New balance: ${self.balance}")
        self.save()

    def buy_shares(self, symbol: str, quantity: int, rationale: str) -> str:
        """ Buy shares of a stock if sufficient funds are available. """
        price = get_share_price(symbol)
        buy_price = price * (1 + SPREAD)
        total_cost = buy_price * quantity
        
        if total_cost > self.balance:
            raise ValueError("Insufficient funds to buy shares.")
        elif price==0:
            raise ValueError(f"Unrecognized symbol {symbol}")
        
        # Update holdings
        self.holdings[symbol] = self.holdings.get(symbol, 0) + quantity
        timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        # Record transaction
        transaction = Transaction(symbol=symbol, quantity=quantity, price=buy_price, timestamp=timestamp, rationale=rationale)
        self.transactions.append(transaction)
        
        # Update balance
        self.balance -= total_cost
        self.save()
        write_log(self.name, "account", f"Bought {quantity} of {symbol}")
        return "Completed. Latest details:\n" + self.report()

    def sell_shares(self, symbol: str, quantity: int, rationale: str) -> str:
        """ Sell shares of a stock if the user has enough shares. """
        if self.holdings.get(symbol, 0) < quantity:
            raise ValueError(f"Cannot sell {quantity} shares of {symbol}. Not enough shares held.")
        
        price = get_share_price(symbol)
        sell_price = price * (1 - SPREAD)
        total_proceeds = sell_price * quantity
        
        # Update holdings
        self.holdings[symbol] -= quantity
        
        # If shares are completely sold, remove from holdings
        if self.holdings[symbol] == 0:
            del self.holdings[symbol]
        timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        # Record transaction
        transaction = Transaction(symbol=symbol, quantity=-quantity, price=sell_price, timestamp=timestamp, rationale=rationale)  # negative quantity for sell
        self.transactions.append(transaction)

        # Update balance
        self.balance += total_proceeds
        self.save()
        write_log(self.name, "account", f"Sold {quantity} of {symbol}")
        return "Completed. Latest details:\n" + self.report()

    def calculate_portfolio_value(self):
        """ Calculate the total value of the user's portfolio. """
        total_value = self.balance
        for symbol, quantity in self.holdings.items():
            total_value += get_share_price(symbol) * quantity
        return total_value

    def calculate_profit_loss(self, portfolio_value: float):
        """ Calculate profit or loss from the initial spend. """
        initial_spend = sum(transaction.total() for transaction in self.transactions)
        return portfolio_value - initial_spend - self.balance

    def get_holdings(self):
        """ Report the current holdings of the user. """
        return self.holdings

    def get_profit_loss(self):
        """ Report the user's profit or loss at any point in time. """
        return self.calculate_profit_loss()

    def list_transactions(self):
        """ List all transactions made by the user. """
        return [transaction.model_dump() for transaction in self.transactions]
    
    def report(self) -> str:
        """ Return a json string representing the account.  """
        portfolio_value = self.calculate_portfolio_value()
        self.portfolio_value_time_series.append((datetime.now().strftime("%Y-%m-%d %H:%M:%S"), portfolio_value))
        self.save()
        pnl = self.calculate_profit_loss(portfolio_value)
        data = self.model_dump()
        data["total_portfolio_value"] = portfolio_value
        data["total_profit_loss"] = pnl
        write_log(self.name, "account", f"Retrieved account details")
        return json.dumps(data)
    
    def get_strategy(self) -> str:
        """ Return the strategy of the account """
        write_log(self.name, "account", f"Retrieved strategy")
        return self.strategy
    
    def change_strategy(self, strategy: str) -> str:
        """ At your discretion, if you choose to, call this to change your investment strategy for the future """
        self.strategy = strategy
        self.save()
        write_log(self.name, "account", f"Changed strategy")
        return "Changed strategy"

// Port of 6_mcp/accounts.py — the business logic of a trading account (written for the course by the Week 3 CrewAI
// engineering team): cash balance, holdings, transactions with a 0.2% spread, a report and an investment strategy.
// Differences: interfaces replace the Pydantic models, methods are camelCase (buy_shares -> buyShares), and the methods
// that need a share price are async because getSharePrice may call Polygon. The stored JSON keeps the Python field
// names (portfolio_value_time_series, ...), so accounts.db looks the same as the course's.
import { readAccount, writeAccount, writeLog } from "./database.js";
import { getSharePrice } from "./market.js";

const INITIAL_BALANCE = 10_000.0;
const SPREAD = 0.002;

export interface Transaction {
  symbol: string;
  quantity: number; // negative for a sale
  price: number;
  timestamp: string;
  rationale: string;
}

export interface AccountData {
  name: string;
  balance: number;
  strategy: string;
  holdings: Record<string, number>;
  transactions: Transaction[];
  portfolio_value_time_series: [string, number][];
}

/** "2025-05-20 14:03:07" in local time, like datetime.now().strftime("%Y-%m-%d %H:%M:%S"). */
export function timestampNow(d = new Date()): string {
  const p = (n: number) => String(n).padStart(2, "0");
  return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())} ${p(d.getHours())}:${p(d.getMinutes())}:${p(d.getSeconds())}`;
}

const total = (t: Transaction) => t.quantity * t.price;

export class Account implements AccountData {
  name = "";
  balance = INITIAL_BALANCE;
  strategy = "";
  holdings: Record<string, number> = {};
  transactions: Transaction[] = [];
  portfolio_value_time_series: [string, number][] = [];

  private constructor(fields: AccountData) {
    Object.assign(this, fields);
  }

  /** Load an account, creating it with $10,000 if it does not exist yet. */
  static get(name: string): Account {
    let fields = readAccount(name.toLowerCase()) as AccountData | null;
    if (!fields) {
      fields = { name: name.toLowerCase(), balance: INITIAL_BALANCE, strategy: "", holdings: {}, transactions: [], portfolio_value_time_series: [] };
      writeAccount(name, fields);
    }
    return new Account(fields);
  }

  /** The plain data, like Pydantic's model_dump(). */
  data(): AccountData {
    const { name, balance, strategy, holdings, transactions, portfolio_value_time_series } = this;
    return { name, balance, strategy, holdings, transactions, portfolio_value_time_series };
  }

  save(): void {
    writeAccount(this.name.toLowerCase(), this.data());
  }

  reset(strategy: string): void {
    this.balance = INITIAL_BALANCE;
    this.strategy = strategy;
    this.holdings = {};
    this.transactions = [];
    this.portfolio_value_time_series = [];
    this.save();
  }

  deposit(amount: number): void {
    if (amount <= 0) throw new Error("Deposit amount must be positive.");
    this.balance += amount;
    console.log(`Deposited $${amount}. New balance: $${this.balance}`);
    this.save();
  }

  withdraw(amount: number): void {
    if (amount > this.balance) throw new Error("Insufficient funds for withdrawal.");
    this.balance -= amount;
    console.log(`Withdrew $${amount}. New balance: $${this.balance}`);
    this.save();
  }

  /** Buy shares at the current price plus the spread, if there is enough cash. */
  async buyShares(symbol: string, quantity: number, rationale: string): Promise<string> {
    const price = await getSharePrice(symbol);
    const buyPrice = price * (1 + SPREAD);
    const totalCost = buyPrice * quantity;

    if (totalCost > this.balance) throw new Error("Insufficient funds to buy shares.");
    else if (price === 0) throw new Error(`Unrecognized symbol ${symbol}`);

    this.holdings[symbol] = (this.holdings[symbol] ?? 0) + quantity;
    this.transactions.push({ symbol, quantity, price: buyPrice, timestamp: timestampNow(), rationale });
    this.balance -= totalCost;
    this.save();
    writeLog(this.name, "account", `Bought ${quantity} of ${symbol}`);
    return "Completed. Latest details:\n" + (await this.report());
  }

  /** Sell shares at the current price minus the spread, if enough are held. */
  async sellShares(symbol: string, quantity: number, rationale: string): Promise<string> {
    if ((this.holdings[symbol] ?? 0) < quantity) throw new Error(`Cannot sell ${quantity} shares of ${symbol}. Not enough shares held.`);

    const price = await getSharePrice(symbol);
    const sellPrice = price * (1 - SPREAD);
    const totalProceeds = sellPrice * quantity;

    this.holdings[symbol] -= quantity;
    if (this.holdings[symbol] === 0) delete this.holdings[symbol]; // sold out: remove from holdings
    this.transactions.push({ symbol, quantity: -quantity, price: sellPrice, timestamp: timestampNow(), rationale });
    this.balance += totalProceeds;
    this.save();
    writeLog(this.name, "account", `Sold ${quantity} of ${symbol}`);
    return "Completed. Latest details:\n" + (await this.report());
  }

  /** Cash plus every holding at today's price. */
  async calculatePortfolioValue(): Promise<number> {
    let totalValue = this.balance;
    for (const [symbol, quantity] of Object.entries(this.holdings)) totalValue += (await getSharePrice(symbol)) * quantity;
    return totalValue;
  }

  /** Profit or loss, computed exactly as in the course. */
  calculateProfitLoss(portfolioValue: number): number {
    const initialSpend = this.transactions.reduce((sum, t) => sum + total(t), 0);
    return portfolioValue - initialSpend - this.balance;
  }

  getHoldings(): Record<string, number> {
    return this.holdings;
  }

  listTransactions(): Transaction[] {
    return this.transactions.map((t) => ({ ...t }));
  }

  /** A JSON string of the account plus total value and profit/loss; also adds a point to the value time series. */
  async report(): Promise<string> {
    const portfolioValue = await this.calculatePortfolioValue();
    this.portfolio_value_time_series.push([timestampNow(), portfolioValue]);
    this.save();
    const pnl = this.calculateProfitLoss(portfolioValue);
    const data = { ...this.data(), total_portfolio_value: portfolioValue, total_profit_loss: pnl };
    writeLog(this.name, "account", "Retrieved account details");
    return JSON.stringify(data);
  }

  getStrategy(): string {
    writeLog(this.name, "account", "Retrieved strategy");
    return this.strategy;
  }

  /** At your discretion, if you choose to, call this to change your investment strategy for the future. */
  changeStrategy(strategy: string): string {
    this.strategy = strategy;
    this.save();
    writeLog(this.name, "account", "Changed strategy");
    return "Changed strategy";
  }
}

The server (lecture 8, 03:34–06:29)

Writing a server is boilerplate: create FastMCP("accounts_server"), decorate functions with @mcp.tool(), and the ordinary docstrings become the tool and argument descriptions. Each tool just calls the business logic. @mcp.resource("accounts://accounts_server/{name}") gives "a sort of" read-only URL to the account's report. The last line runs it over stdio.

// Port of 6_mcp/accounts_server.py — our own MCP server: 5 tools and 2 resources over stdio, wrapping accounts.ts.
// Differences: FastMCP's @mcp.tool() decorators become server.registerTool(name, {description, inputSchema}, handler)
// from the official TypeScript SDK (@modelcontextprotocol/server v2); the argument docs move into zod .describe() calls,
// and each handler returns MCP "content" (text) explicitly where FastMCP converts return values for you.
// Never console.log in a stdio server: stdout carries the protocol messages. (Errors thrown by a handler reach the
// client as a tool error, as FastMCP does with a raised ValueError.)
import { McpServer, ResourceTemplate } from "@modelcontextprotocol/server";
import { serveStdio } from "@modelcontextprotocol/server/stdio";
import { z } from "zod";
import { Account } from "./accounts.js";

const text = (value: unknown) => ({
  content: [{ type: "text" as const, text: typeof value === "string" ? value : JSON.stringify(value) }],
});

function createServer(): McpServer {
  const mcp = new McpServer({ name: "accounts_server", version: "1.0.0" });

  mcp.registerTool(
    "get_balance",
    {
      description: "Get the cash balance of the given account name.",
      inputSchema: z.object({ name: z.string().describe("The name of the account holder") }),
    },
    async ({ name }) => text(Account.get(name).balance),
  );

  mcp.registerTool(
    "get_holdings",
    {
      description: "Get the holdings of the given account name.",
      inputSchema: z.object({ name: z.string().describe("The name of the account holder") }),
    },
    async ({ name }) => text(Account.get(name).holdings),
  );

  mcp.registerTool(
    "buy_shares",
    {
      description: "Buy shares of a stock.",
      inputSchema: z.object({
        name: z.string().describe("The name of the account holder"),
        symbol: z.string().describe("The symbol of the stock"),
        quantity: z.number().int().describe("The quantity of shares to buy"),
        rationale: z.string().describe("The rationale for the purchase and fit with the account's strategy"),
      }),
    },
    async ({ name, symbol, quantity, rationale }) => text(await Account.get(name).buyShares(symbol, quantity, rationale)),
  );

  mcp.registerTool(
    "sell_shares",
    {
      description: "Sell shares of a stock.",
      inputSchema: z.object({
        name: z.string().describe("The name of the account holder"),
        symbol: z.string().describe("The symbol of the stock"),
        quantity: z.number().int().describe("The quantity of shares to sell"),
        rationale: z.string().describe("The rationale for the sale and fit with the account's strategy"),
      }),
    },
    async ({ name, symbol, quantity, rationale }) => text(await Account.get(name).sellShares(symbol, quantity, rationale)),
  );

  mcp.registerTool(
    "change_strategy",
    {
      description: "At your discretion, if you choose to, call this to change your investment strategy for the future.",
      inputSchema: z.object({
        name: z.string().describe("The name of the account holder"),
        strategy: z.string().describe("The new strategy for the account"),
      }),
    },
    async ({ name, strategy }) => text(Account.get(name).changeStrategy(strategy)),
  );

  // Resources: read-only context the client can fetch by URI (the {name} part is a template variable).
  mcp.registerResource(
    "read_account_resource",
    new ResourceTemplate("accounts://accounts_server/{name}", { list: undefined }),
    { mimeType: "text/plain" },
    async (uri, { name }) => ({ contents: [{ uri: uri.href, text: await Account.get(String(name).toLowerCase()).report() }] }),
  );

  mcp.registerResource(
    "read_strategy_resource",
    new ResourceTemplate("accounts://strategy/{name}", { list: undefined }),
    { mimeType: "text/plain" },
    async (uri, { name }) => ({ contents: [{ uri: uri.href, text: Account.get(String(name).toLowerCase()).getStrategy() }] }),
  );

  return mcp;
}

// Port of 6_mcp/accounts_client.py — a hand-written MCP client for our accounts server.
// Each function starts the server as a child process, opens a session, does one thing and closes it again,
// exactly like the Python version's "async with stdio_client(...) / ClientSession(...)" blocks.
// Differences: Client + StdioClientTransport from @modelcontextprotocol/client v2 replace ClientSession + stdio_client,
// a small withSession() helper replaces the repeated "async with" blocks, and tool() from @openai/agents with a plain
// JSON schema replaces FunctionTool(params_json_schema=...). The server runs with "node --import tsx" (the "uv run" of TS).
import { Client } from "@modelcontextprotocol/client";
import { StdioClientTransport } from "@modelcontextprotocol/client/stdio";
import { tool } from "@openai/agents";

const params = { command: process.execPath, args: ["--import", "tsx", "src/accounts_server.ts"] };

async function withSession<T>(work: (client: Client) => Promise<T>): Promise<T> {
  const client = new Client({ name: "accounts_client", version: "1.0.0" });
  await client.connect(new StdioClientTransport(params)); // spawns the server and runs the initialize handshake
  try {
    return await work(client);
  } finally {
    await client.close();
  }
}

export async function listAccountsTools() {
  return withSession(async (client) => (await client.listTools()).tools);
}

export async function callAccountsTool(toolName: string, toolArgs: Record<string, unknown>) {
  return withSession((client) => client.callTool({ name: toolName, arguments: toolArgs }));
}

async function readText(uri: string): Promise<string> {
  return withSession(async (client) => {
    const result = await client.readResource({ uri });
    const first = result.contents[0];
    return "text" in first ? first.text : "";
  });
}

export const readAccountsResource = (name: string) => readText(`accounts://accounts_server/${name}`);
export const readStrategyResource = (name: string) => readText(`accounts://strategy/${name}`);

/** Turn each MCP tool into an Agents SDK function tool whose execute() calls the MCP server. */
export async function getAccountsToolsOpenai() {
  const openaiTools = [];
  for (const t of await listAccountsTools()) {
    const schema = { ...t.inputSchema, additionalProperties: false } as any;
    openaiTools.push(
      tool({
        name: t.name,
        description: t.description ?? "",
        parameters: schema,
        strict: true,
        execute: async (args) => JSON.stringify(await callAccountsTool(t.name, args as Record<string, unknown>)),
      }),
    );
  }
  return openaiTools;
}

Common mistakes

  • Running from the wrong folder: uv run accounts_server.py and accounts.db are relative to 6_mcp/.
  • Printing from a stdio server: output on standard output corrupts the protocol; log to standard error.
  • Building a server for your own private tool: a function tool is simpler.
  • Missing docstrings or descriptions: they are all the model knows about your tools.
  • Forgetting initialize() in a hand-written client: the session must shake hands first.
Recap
  • Build an MCP server to share tools and resources; for private tools, use a function tool.
  • FastMCP plus @mcp.tool() and @mcp.resource() (TypeScript: McpServer.registerTool and registerResource) wrap existing code.
  • An agent uses our server exactly like someone else's: MCPServerStdio with the command line.
  • A client starts the server, initializes a session, then lists tools, calls tools and reads resources.
