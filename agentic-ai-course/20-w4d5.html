What this day builds

The notebook Sidekick becomes a real app in three files: sidekick_tools.py (all the tools), sidekick.py (a Sidekick class with the worker, the evaluator and the graph) and app.py (the Gradio UI, one Sidekick per browser session). The agenda is "Tools, tools, tools": web search, push notifications, files, Wikipedia and a Python REPL. The presenter has used it "for real commercial work": "see it as a canvas".

TypeScript: src/sidekick/ holds the tools and the class, server/index.ts is a Hono API that keeps one Sidekick per browser tab, and web/ is the React UI. Run pnpm build then pnpm api, and open http://localhost:8790.

// ---- file tools (Python: FileManagementToolkit(root_dir="sandbox").get_tools()) ----
const ROOT = path.resolve("sandbox"); // relative to where you start the program, like the Python app

/** Resolve a path inside sandbox/ and refuse anything that would escape it (e.g. "../.env"). */
function inSandbox(p: string): string {
  const full = path.resolve(ROOT, p);
  if (full !== ROOT && !full.startsWith(ROOT + path.sep)) throw new Error(`Access denied: ${p} is outside the sandbox`);
  return full;
}
const pathArg = (description: string) => z.string().describe(description);

/** Wrap a file operation as a tool; errors (missing file, path outside the sandbox) go back to the model as text. */
function fileTool<S extends z.ZodObject>(name: string, description: string, schema: S, run: (args: z.infer<S>) => Promise<string>) {
  const safeRun = async (args: z.infer<S>) => {
    try {
      return await run(args);
    } catch (e) {
      return `Error: ${(e as Error).message}`;
    }
  };
  return tool(safeRun, { name, description, schema });
}

function get_file_tools(): StructuredToolInterface[] {
  const copy_file = fileTool("copy_file", "Create a copy of a file in a specified location",
    z.object({ source_path: pathArg("Path of the file to copy"), destination_path: pathArg("Path to save the copied file") }),
    async ({ source_path, destination_path }) => {
      await fs.copyFile(inSandbox(source_path), inSandbox(destination_path));
      return `File copied successfully from ${source_path} to ${destination_path}.`;
    });

  const file_delete = fileTool("file_delete", "Delete a file", z.object({ file_path: pathArg("Path of the file to delete") }),
    async ({ file_path }) => {
      await fs.unlink(inSandbox(file_path));
      return `File deleted successfully: ${file_path}.`;
    });

  const file_search = fileTool("file_search", "Recursively search for files in a subdirectory that match the regex pattern",
    z.object({ dir_path: pathArg("Subdirectory to search in.").default("."), pattern: z.string().describe("Unix shell regex, where * matches everything.") }),
    async ({ dir_path, pattern }) => {
      const entries = await fs.readdir(inSandbox(dir_path), { recursive: true });
      // turn the shell-style pattern (*.md) into a regular expression (^.*\.md$)
      const re = new RegExp("^" + pattern.replace(/[.+^${}()|[\]\\]/g, "\\$&").replace(/\*/g, ".*").replace(/\?/g, ".") + "$");
      const hits = entries.filter((e) => re.test(path.basename(e)));
      return hits.length ? hits.join("\n") : `No files found for pattern ${pattern} in directory ${dir_path}`;
    });

  const move_file = fileTool("move_file", "Move or rename a file from one location to another",
    z.object({ source_path: pathArg("Path of the file to move"), destination_path: pathArg("New path for the moved file") }),
    async ({ source_path, destination_path }) => {
      await fs.rename(inSandbox(source_path), inSandbox(destination_path));
      return `File moved successfully from ${source_path} to ${destination_path}.`;
    });

  const read_file = fileTool("read_file", "Read file from disk", z.object({ file_path: pathArg("name of file") }),
    ({ file_path }) => fs.readFile(inSandbox(file_path), "utf8"));

  const write_file = fileTool("write_file", "Write file to disk",
    z.object({ file_path: pathArg("name of file"), text: z.string().describe("text to write to file"), append: z.boolean().default(false).describe("Whether to append to an existing file.") }),
    async ({ file_path, text, append }) => {
      const full = inSandbox(file_path);
      await fs.mkdir(path.dirname(full), { recursive: true });
      if (append) await fs.appendFile(full, text, "utf8");
      else await fs.writeFile(full, text, "utf8");
      return `File written successfully to ${file_path}.`;
    });

  const list_directory = fileTool("list_directory", "List files and directories in a specified folder",
    z.object({ dir_path: pathArg("Subdirectory to list.").default(".") }),
    async ({ dir_path }) => (await fs.readdir(inSandbox(dir_path))).join("\n") || `No files found in directory ${dir_path}`);

  return [copy_file, file_delete, file_search, move_file, read_file, write_file, list_directory];
}

// ---- Wikipedia (Python: WikipediaQueryRun(api_wrapper=WikipediaAPIWrapper())) via the public REST API ----
async function wikipediaRun(query: string): Promise<string> {
  const headers = { "User-Agent": "agentic-ai-week4-ts/0.1 (course exercise)" }; // Wikipedia asks API clients to identify themselves
  const search = new URL("https://en.wikipedia.org/w/api.php");
  search.search = new URLSearchParams({ action: "query", list: "search", srsearch: query, srlimit: "3", format: "json" }).toString();
  const hits = ((await (await fetch(search, { headers })).json()) as { query?: { search: { title: string }[] } }).query?.search ?? [];
  const pages = await Promise.all(hits.map(async ({ title }) => {
    const res = await fetch(`https://en.wikipedia.org/api/rest_v1/page/summary/${encodeURIComponent(title)}`, { headers });
    const { extract } = (await res.json()) as { extract?: string };
    return `Page: ${title}\nSummary: ${extract ?? ""}`; // the same "Page: / Summary:" layout as the Python wrapper
  }));
  return pages.length ? pages.join("\n\n").slice(0, 4000) : "No good Wikipedia Search Result was found";
}

// ---- Python (Python: PythonREPLTool(), which runs code on your own machine without any sandbox) ----
/** Run Python in a throw-away Docker container instead: no network, 256 MB of memory, a time limit. */
export function runPythonInDocker(code: string, timeoutSeconds = 30): Promise<string> {
  return new Promise((resolve) => {
    const child = execFile(
      "docker",
      ["run", "--rm", "-i", "--network", "none", "--memory", "256m", "python:3.12-slim", "python", "-"],
      { timeout: timeoutSeconds * 1000, maxBuffer: 1024 * 1024 },
      (err, stdout, stderr) => resolve(err && !stdout ? `Error: ${stderr || err.message}` : stdout + (stderr ? `\n[stderr]\n${stderr}` : "")),
    );
    child.stdin?.end(code); // the program goes in on standard input ("python -" reads it from there)
  });
}

The Sidekick class (lectures 19–20)

Lab 4's code moves into a class. One wrinkle: __init__ cannot be async, but setting up the browser needs await. So there is a separate setup(): "we can first instantiate it and then call this setup asynchronously." The notebook was where he could "prototype, iterate, perfect your prompts, and then move to a module"; AI engineering "is much more experimental by nature".

from typing import Annotated
from typing_extensions import TypedDict
from langgraph.graph import StateGraph, START, END
from langgraph.graph.message import add_messages
from dotenv import load_dotenv
from langgraph.prebuilt import ToolNode
from langchain_openai import ChatOpenAI
from langgraph.checkpoint.memory import MemorySaver
from langchain_core.messages import AIMessage, HumanMessage, SystemMessage
from typing import List, Any, Optional, Dict
from pydantic import BaseModel, Field
from sidekick_tools import playwright_tools, other_tools
import uuid
import asyncio
from datetime import datetime

load_dotenv(override=True)


class State(TypedDict):
    messages: Annotated[List[Any], add_messages]
    success_criteria: str
    feedback_on_work: Optional[str]
    success_criteria_met: bool
    user_input_needed: bool


class EvaluatorOutput(BaseModel):
    feedback: str = Field(description="Feedback on the assistant's response")
    success_criteria_met: bool = Field(description="Whether the success criteria have been met")
    user_input_needed: bool = Field(
        description="True if more input is needed from the user, or clarifications, or the assistant is stuck"
    )


class Sidekick:
    def __init__(self):
        self.worker_llm_with_tools = None
        self.evaluator_llm_with_output = None
        self.tools = None
        self.llm_with_tools = None
        self.graph = None
        self.sidekick_id = str(uuid.uuid4())
        self.memory = MemorySaver()
        self.browser = None
        self.playwright = None

    async def setup(self):
        self.tools, self.browser, self.playwright = await playwright_tools()
        self.tools += await other_tools()
        worker_llm = ChatOpenAI(model="gpt-4o-mini")
        self.worker_llm_with_tools = worker_llm.bind_tools(self.tools)
        evaluator_llm = ChatOpenAI(model="gpt-4o-mini")
        self.evaluator_llm_with_output = evaluator_llm.with_structured_output(EvaluatorOutput)
        await self.build_graph()

    def worker(self, state: State) -> Dict[str, Any]:
        system_message = f"""You are a helpful assistant that can use tools to complete tasks.
    You keep working on a task until either you have a question or clarification for the user, or the success criteria is met.
    You have many tools to help you, including tools to browse the internet, navigating and retrieving web pages.
    You have a tool to run python code, but note that you would need to include a print() statement if you wanted to receive output.
    The current date and time is {datetime.now().strftime("%Y-%m-%d %H:%M:%S")}

    This is the success criteria:
    {state["success_criteria"]}
    You should reply either with a question for the user about this assignment, or with your final response.
    If you have a question for the user, you need to reply by clearly stating your question. An example might be:

    Question: please clarify whether you want a summary or a detailed answer

    If you've finished, reply with the final answer, and don't ask a question; simply reply with the answer.
    """

        if state.get("feedback_on_work"):
            system_message += f"""
    Previously you thought you completed the assignment, but your reply was rejected because the success criteria was not met.
    Here is the feedback on why this was rejected:
    {state["feedback_on_work"]}
    With this feedback, please continue the assignment, ensuring that you meet the success criteria or have a question for the user."""

        # Add in the system message

        found_system_message = False
        messages = state["messages"]
        for message in messages:
            if isinstance(message, SystemMessage):
                message.content = system_message
                found_system_message = True

        if not found_system_message:
            messages = [SystemMessage(content=system_message)] + messages

        # Invoke the LLM with tools
        response = self.worker_llm_with_tools.invoke(messages)

        # Return updated state
        return {
            "messages": [response],
        }

    def worker_router(self, state: State) -> str:
        last_message = state["messages"][-1]

        if hasattr(last_message, "tool_calls") and last_message.tool_calls:
            return "tools"
        else:
            return "evaluator"

    def format_conversation(self, messages: List[Any]) -> str:
        conversation = "Conversation history:\n\n"
        for message in messages:
            if isinstance(message, HumanMessage):
                conversation += f"User: {message.content}\n"
            elif isinstance(message, AIMessage):
                text = message.content or "[Tools use]"
                conversation += f"Assistant: {text}\n"
        return conversation

    def evaluator(self, state: State) -> State:
        last_response = state["messages"][-1].content

        system_message = """You are an evaluator that determines if a task has been completed successfully by an Assistant.
    Assess the Assistant's last response based on the given criteria. Respond with your feedback, and with your decision on whether the success criteria has been met,
    and whether more input is needed from the user."""

        user_message = f"""You are evaluating a conversation between the User and Assistant. You decide what action to take based on the last response from the Assistant.

    The entire conversation with the assistant, with the user's original request and all replies, is:
    {self.format_conversation(state["messages"])}

    The success criteria for this assignment is:
    {state["success_criteria"]}

    And the final response from the Assistant that you are evaluating is:
    {last_response}

    Respond with your feedback, and decide if the success criteria is met by this response.
    Also, decide if more user input is required, either because the assistant has a question, needs clarification, or seems to be stuck and unable to answer without help.

    The Assistant has access to a tool to write files. If the Assistant says they have written a file, then you can assume they have done so.
    Overall you should give the Assistant the benefit of the doubt if they say they've done something. But you should reject if you feel that more work should go into this.

    """
        if state["feedback_on_work"]:
            user_message += f"Also, note that in a prior attempt from the Assistant, you provided this feedback: {state['feedback_on_work']}\n"
            user_message += "If you're seeing the Assistant repeating the same mistakes, then consider responding that user input is required."

        evaluator_messages = [
            SystemMessage(content=system_message),
            HumanMessage(content=user_message),
        ]

        eval_result = self.evaluator_llm_with_output.invoke(evaluator_messages)
        new_state = {
            "messages": [
                {
                    "role": "assistant",
                    "content": f"Evaluator Feedback on this answer: {eval_result.feedback}",
                }
            ],
            "feedback_on_work": eval_result.feedback,
            "success_criteria_met": eval_result.success_criteria_met,
            "user_input_needed": eval_result.user_input_needed,
        }
        return new_state

    def route_based_on_evaluation(self, state: State) -> str:
        if state["success_criteria_met"] or state["user_input_needed"]:
            return "END"
        else:
            return "worker"

    async def build_graph(self):
        # Set up Graph Builder with State
        graph_builder = StateGraph(State)

        # Add nodes
        graph_builder.add_node("worker", self.worker)
        graph_builder.add_node("tools", ToolNode(tools=self.tools))
        graph_builder.add_node("evaluator", self.evaluator)

        # Add edges
        graph_builder.add_conditional_edges(
            "worker", self.worker_router, {"tools": "tools", "evaluator": "evaluator"}
        )
        graph_builder.add_edge("tools", "worker")
        graph_builder.add_conditional_edges(
            "evaluator", self.route_based_on_evaluation, {"worker": "worker", "END": END}
        )
        graph_builder.add_edge(START, "worker")

        # Compile the graph
        self.graph = graph_builder.compile(checkpointer=self.memory)

    async def run_superstep(self, message, success_criteria, history):
        config = {"configurable": {"thread_id": self.sidekick_id}}

        state = {
            "messages": message,
            "success_criteria": success_criteria or "The answer should be clear and accurate",
            "feedback_on_work": None,
            "success_criteria_met": False,
            "user_input_needed": False,
        }
        result = await self.graph.ainvoke(state, config=config)
        user = {"role": "user", "content": message}
        reply = {"role": "assistant", "content": result["messages"][-2].content}
        feedback = {"role": "assistant", "content": result["messages"][-1].content}
        return history + [user, reply, feedback]

    def cleanup(self):
        if self.browser:
            try:
                loop = asyncio.get_running_loop()
                loop.create_task(self.browser.close())
                if self.playwright:
                    loop.create_task(self.playwright.stop())
            except RuntimeError:
                # If no loop is running, do a direct run
                asyncio.run(self.browser.close())
                if self.playwright:
                    asyncio.run(self.playwright.stop())

export type ChatMessage = { role: "user" | "assistant"; content: string };

function buildWorkerLLM(tools: StructuredToolInterface[]) {
  return makeLLM().bindTools(tools);
}
function buildEvaluatorLLM() {
  return makeLLM().withStructuredOutput(EvaluatorOutput);
}

export class Sidekick {
  private worker_llm_with_tools?: ReturnType<typeof buildWorkerLLM>;
  private evaluator_llm_with_output?: ReturnType<typeof buildEvaluatorLLM>;
  private tools: StructuredToolInterface[] = [];
  private graph?: Awaited<ReturnType<Sidekick["buildGraph"]>>;
  readonly sidekick_id = randomUUID(); // also the checkpointer's thread_id: one conversation per Sidekick
  private memory = new MemorySaver();
  private browser?: Browser;

  async setup() {
    const { tools, browser } = await playwright_tools();
    this.tools = [...tools, ...(await other_tools())];
    this.browser = browser;
    this.worker_llm_with_tools = buildWorkerLLM(this.tools);
    this.evaluator_llm_with_output = buildEvaluatorLLM();
    this.graph = await this.buildGraph();
  }

  // Arrow-function properties keep `this` bound when LangGraph calls them as nodes (Python's self.worker does this for free)
  worker = async (state: StateT) => {
    // the date is local time as "YYYY-MM-DD HH:MM:SS" (the sv-SE locale happens to use exactly Python's strftime format)
    let system_message = `You are a helpful assistant that can use tools to complete tasks.
    You keep working on a task until either you have a question or clarification for the user, or the success criteria is met.
    You have many tools to help you, including tools to browse the internet, navigating and retrieving web pages.
    You have a tool to run python code, but note that you would need to include a print() statement if you wanted to receive output.
    The current date and time is ${new Date().toLocaleString("sv-SE")}

    This is the success criteria:
    ${state.success_criteria}
    You should reply either with a question for the user about this assignment, or with your final response.
    If you have a question for the user, you need to reply by clearly stating your question. An example might be:

    Question: please clarify whether you want a summary or a detailed answer

    If you've finished, reply with the final answer, and don't ask a question; simply reply with the answer.
    `;

    if (state.feedback_on_work) {
      system_message += `
    Previously you thought you completed the assignment, but your reply was rejected because the success criteria was not met.
    Here is the feedback on why this was rejected:
    ${state.feedback_on_work}
    With this feedback, please continue the assignment, ensuring that you meet the success criteria or have a question for the user.`;
    }

    // Add in the system message
    const found_system_message = state.messages.some((m) => SystemMessage.isInstance(m));
    const messages = found_system_message
      ? state.messages.map((m) => (SystemMessage.isInstance(m) ? new SystemMessage(system_message) : m))
      : [new SystemMessage(system_message), ...state.messages];

    // Invoke the LLM with tools
    const response = await this.worker_llm_with_tools!.invoke(messages);

    // Return updated state
    return { messages: [response] };
  };

  workerRouter = (state: StateT): "tools" | "evaluator" => {
    const last_message = state.messages.at(-1);
    return AIMessage.isInstance(last_message) && last_message.tool_calls?.length ? "tools" : "evaluator";
  };

  formatConversation(messages: BaseMessage[]): string {
    let conversation = "Conversation history:\n\n";
    for (const message of messages) {
      if (HumanMessage.isInstance(message)) conversation += `User: ${message.content}\n`;
      else if (AIMessage.isInstance(message)) conversation += `Assistant: ${message.content || "[Tools use]"}\n`;
    }
    return conversation;
  }

  evaluator = async (state: StateT): Promise<typeof State.Update> => {
    const last_response = state.messages.at(-1)!.content;

    const system_message = `You are an evaluator that determines if a task has been completed successfully by an Assistant.
    Assess the Assistant's last response based on the given criteria. Respond with your feedback, and with your decision on whether the success criteria has been met,
    and whether more input is needed from the user.`;

    let user_message = `You are evaluating a conversation between the User and Assistant. You decide what action to take based on the last response from the Assistant.

    The entire conversation with the assistant, with the user's original request and all replies, is:
    ${this.formatConversation(state.messages)}

    The success criteria for this assignment is:
    ${state.success_criteria}

    And the final response from the Assistant that you are evaluating is:
    ${last_response}

    Respond with your feedback, and decide if the success criteria is met by this response.
    Also, decide if more user input is required, either because the assistant has a question, needs clarification, or seems to be stuck and unable to answer without help.

    The Assistant has access to a tool to write files. If the Assistant says they have written a file, then you can assume they have done so.
    Overall you should give the Assistant the benefit of the doubt if they say they've done something. But you should reject if you feel that more work should go into this.

    `;
    if (state.feedback_on_work) {
      user_message += `Also, note that in a prior attempt from the Assistant, you provided this feedback: ${state.feedback_on_work}\n`;
      user_message += "If you're seeing the Assistant repeating the same mistakes, then consider responding that user input is required.";
    }

    const eval_result = await this.evaluator_llm_with_output!.invoke([new SystemMessage(system_message), new HumanMessage(user_message)]);
    return {
      messages: [{ role: "assistant", content: `Evaluator Feedback on this answer: ${eval_result.feedback}` }],
      feedback_on_work: eval_result.feedback,
      success_criteria_met: eval_result.success_criteria_met,
      user_input_needed: eval_result.user_input_needed,
    };
  };

  routeBasedOnEvaluation = (state: StateT): "worker" | "END" =>
    state.success_criteria_met || state.user_input_needed ? "END" : "worker";

  async buildGraph() {
    // Set up Graph Builder with State, add nodes, add edges, compile
    return new StateGraph(State)
      .addNode("worker", this.worker)
      .addNode("tools", new ToolNode(this.tools))
      .addNode("evaluator", this.evaluator)
      .addConditionalEdges("worker", this.workerRouter, { tools: "tools", evaluator: "evaluator" })
      .addEdge("tools", "worker")
      .addConditionalEdges("evaluator", this.routeBasedOnEvaluation, { worker: "worker", END: END })
      .addEdge(START, "worker")
      .compile({ checkpointer: this.memory });
  }

  /** One super-step: the user's message goes in, the worker/evaluator loop runs, the chat history comes back
   *  with three more entries: the user's message, the worker's reply and the evaluator's feedback. */
  async runSuperstep(message: string, success_criteria: string, history: ChatMessage[]): Promise<ChatMessage[]> {
    const config = { configurable: { thread_id: this.sidekick_id } };

    const state = {
      messages: [{ role: "user", content: message }],
      success_criteria: success_criteria || "The answer should be clear and accurate",
      feedback_on_work: null,
      success_criteria_met: false,
      user_input_needed: false,
    };
    const result = await this.graph!.invoke(state, config);
    const user: ChatMessage = { role: "user", content: message };
    const reply: ChatMessage = { role: "assistant", content: String(result.messages.at(-2)!.content) };
    const feedback: ChatMessage = { role: "assistant", content: String(result.messages.at(-1)!.content) };
    return [...history, user, reply, feedback];
  }

  /** Close Chromium. Python needs event-loop juggling here; in Node an await is enough. */
  async cleanup() {
    await this.browser?.close().catch(() => {});
  }
}

One Sidekick per user (lecture 21)

Gradio wires events to functions: "these callbacks are called back to your actual server". ui.load fires when a page loads and calls setup, which returns a new Sidekick stored in gr.State, "associated with a particular user of this screen". delete_callback=free_resources is "how I'm hoping to clean up this Playwright and Chromium browser". He confesses that in an earlier course "everyone will be sharing the same variables"; this time each user has "their own separate session".

// Week 4 Day 5 — the Sidekick API behind the React app (replaces the Gradio app app.py)
// Gradio keeps one Sidekick per browser tab in gr.State; here the browser sends a threadId and the server keeps
// one Sidekick per threadId in a Map. Each Sidekick has its own Chromium and its own conversation memory.
//   POST /api/run   {message, successCriteria, history, threadId} → {history} (user message, reply, evaluator feedback added)
//   POST /api/reset {threadId}                                     → {threadId} (a fresh Sidekick's id)
import { Hono } from "hono";
import { serve } from "@hono/node-server";
import { serveStatic } from "@hono/node-server/serve-static";
import { randomUUID } from "node:crypto";
import { Sidekick, type ChatMessage } from "../src/sidekick/sidekick.js";

type Session = { sidekick: Promise<Sidekick>; lastUsed: number };
const sessions = new Map<string, Session>();

/** Python: async def setup(): sidekick = Sidekick(); await sidekick.setup() — done the first time a thread is seen */
function getSidekick(threadId: string): Promise<Sidekick> {
  let session = sessions.get(threadId);
  if (!session) {
    const sidekick = new Sidekick();
    // store the promise right away, so two quick requests on one thread do not start two browsers
    session = { sidekick: sidekick.setup().then(() => sidekick), lastUsed: Date.now() };
    sessions.set(threadId, session);
  }
  session.lastUsed = Date.now();
  return session.sidekick;
}

/** Python: free_resources(sidekick), which Gradio calls through gr.State(delete_callback=...) */
async function freeResources(threadId: string) {
  const session = sessions.get(threadId);
  if (!session) return;
  sessions.delete(threadId);
  console.log("Cleaning up", threadId);
  try {
    await (await session.sidekick).cleanup();
  } catch (e) {
    console.log(`Exception during cleanup: ${e}`);
  }
}

const app = new Hono();

// Python: process_message(sidekick, message, success_criteria, history)
app.post("/api/run", async (c) => {
  const { message, successCriteria, history, threadId } = await c.req.json<{ message: string; successCriteria: string; history: ChatMessage[]; threadId: string }>();
  if (!message?.trim() || !threadId) return c.json({ error: "message and threadId are required" }, 400);
  try {
    const sidekick = await getSidekick(threadId);
    return c.json({ history: await sidekick.runSuperstep(message, successCriteria, history ?? []) });
  } catch (e) {
    console.error(e);
    return c.json({ error: String(e) }, 500); // e.g. GraphRecursionError when worker and evaluator loop 25 steps
  }
});

// Python: reset() — throw the old Sidekick away and start a new one with empty memory
app.post("/api/reset", async (c) => {
  const { threadId } = await c.req.json<{ threadId?: string }>().catch(() => ({ threadId: undefined }));
  if (threadId) await freeResources(threadId);
  return c.json({ threadId: randomUUID() });
});

app.use("/*", serveStatic({ root: "./web/dist" })); // production build of the React app (pnpm build)

// Close the browsers of tabs that have been idle for 30 minutes (Gradio does the same when a session expires)
setInterval(() => {
  for (const [id, s] of sessions) if (Date.now() - s.lastUsed > 30 * 60_000) void freeResources(id);
}, 60_000).unref();

// Ctrl+C: close every Chromium before exiting
process.on("SIGINT", async () => {
  await Promise.all([...sessions.keys()].map(freeResources));
  process.exit(0);
});

const port = Number(process.env.PORT ?? 8790);
serve({ fetch: app.fetch, port });
console.log(`Sidekick API on http://localhost:${port}`);

// The Sidekick UI in React. It replaces the Gradio app (app.py):
// gr.Chatbot → the message list, two gr.Textbox → two <input>, "Reset" and "Go!" buttons → <button>,
// gr.State(sidekick) → a threadId the server uses to find this tab's Sidekick.
import { useState } from "react";
import { marked } from "marked";

type ChatMessage = { role: "user" | "assistant"; content: string };

const FEEDBACK_PREFIX = "Evaluator Feedback on this answer:"; // the evaluator node starts its message with this

export function App() {
  const [threadId, setThreadId] = useState<string>(() => crypto.randomUUID());
  const [history, setHistory] = useState<ChatMessage[]>([]);
  const [message, setMessage] = useState("");
  const [successCriteria, setSuccessCriteria] = useState("");
  const [running, setRunning] = useState(false);
  const [error, setError] = useState("");

  // Python: process_message — wired to Go! and to Enter in either textbox, like message.submit / success_criteria.submit
  async function processMessage() {
    if (!message.trim() || running) return;
    setRunning(true);
    setError("");
    try {
      const res = await fetch("/api/run", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ message, successCriteria, history, threadId }),
      });
      const data = (await res.json()) as { history?: ChatMessage[]; error?: string };
      if (data.history) setHistory(data.history);
      else setError(data.error ?? `HTTP ${res.status}`);
    } catch (e) {
      setError(String(e));
    } finally {
      setRunning(false);
    }
  }

  // Python: reset — clear both boxes and the chat, and get a brand-new Sidekick
  async function reset() {
    const res = await fetch("/api/reset", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ threadId }) });
    const data = (await res.json()) as { threadId: string };
    setThreadId(data.threadId);
    setMessage("");
    setSuccessCriteria("");
    setHistory([]);
    setError("");
  }

  const onEnter = (e: React.KeyboardEvent) => e.key === "Enter" && processMessage();

  return (
    <main className="wrap">
      <h2>Sidekick Personal Co-Worker</h2>

      <section className="chatbot" aria-label="Sidekick">
        <span className="label">Sidekick</span>
        {history.map((m, i) => {
          const isFeedback = m.role === "assistant" && m.content.startsWith(FEEDBACK_PREFIX);
          return (
            <div key={i} className={`msg ${m.role}${isFeedback ? " feedback" : ""}`}
              dangerouslySetInnerHTML={{ __html: marked.parse(m.content) as string }} />
          );
        })}
        {running && <div className="msg assistant pending">Sidekick is working… (browsing and tool calls can take a few minutes)</div>}
      </section>

      <div className="group">
        <input id="message" value={message} placeholder="Your request to the Sidekick" onChange={(e) => setMessage(e.target.value)} onKeyDown={onEnter} />
        <input id="criteria" value={successCriteria} placeholder="What are your success criteria?" onChange={(e) => setSuccessCriteria(e.target.value)} onKeyDown={onEnter} />
      </div>

      <div className="row">
        <button id="reset" className="stop" onClick={reset} disabled={running}>Reset</button>
        <button id="go" className="primary" onClick={processMessage} disabled={running}>{running ? "Working…" : "Go!"}</button>
      </div>
      {error && <p className="error">Error: {error}</p>}
    </main>
  );
}

The feedback loop, live (lecture 22)

First, a confession: he had left the brackets off PythonREPLTool and the app crashed. "When things stop working, the first thing you should be suspicious of is what did you change?"

"what is pi * 3?" with no criteria, so the default applies. The answer appears in seconds, but LangSmith shows what happened. The worker ran Python and answered "approximately 9.425". The evaluator rejected it: "should be approximately 9.42477796, not 9.425… rounded the answer too early". The worker retried, sent code with a SyntaxError, fixed it, and answered 9.42477796, which passed. "It all happened in a couple of seconds. So we weren't aware that this syntax error, this feedback was happening and we got to the right answer."

The restaurant. "I'd like to go for dinner tomorrow in a French restaurant in NYC. Please find a great French restaurant and write a report in markdown to dinner.md including the name, address, menu, reviews. Send me a push notification with the restaurant name and phone." A Chromium window opened but stayed blank; the worker used search instead. It wrote dinner.md with Le Bernardin, Balthazar and Daniel, and pushed Le Bernardin's number, (212) 554-1515, which he checked on Google. A follow-up, "Please update the file to only contain information about Le Bernadin…", worked without naming the file: "it remembered that it had been called dinner, which it was getting from LangGraph's checkpointing." The repo's sandbox/dinner.md is that exact final report.

A tip: LLMs are "great at generating markdown"; for PDF reports, "add a tool that just converts markdown to PDF. That's the trick."

Make it your own (lecture 23)

  • Clarifying questions: have it ask three questions before starting, as OpenAI's deep research does.
  • A planning agent that splits the task and delegates. One agent with many tools can lose coherence because "the context is so long", but a planner takes away some freedom to adapt. "There's no right answer."
  • Persistent memory: swap MemorySaver for Day 3's SQLite checkpointer and use a Gradio login's username as the thread ID, so it remembers you.
  • More tools: SQL, Google Calendar, markdown-to-PDF; share them in community_contributions.

Besides the cleanup fix, the repo adds title="Sidekick", changes the placeholder to "Your request to the Sidekick", turns an f-string prompt into a plain string and reformats both files (the video's sidekick.py is 210 lines, the repo's 222). Common mistakes

  • Starting the app from the wrong folder: sandbox is relative, so run uv run app.py from 4_langgraph/ (and the TypeScript server from week4-ts/).
  • No SERPER_API_KEY: the Python tools module builds the Serper wrapper at import time.
  • Forgetting () after a tool class: PythonREPLTool instead of PythonREPLTool() crashes the app.
  • Chromium windows piling up: make sure cleanup really runs; the recorded version never closed them.
  • Trusting the REPL: in Python it runs on your machine with your permissions.
Recap
  • Three modules: tools, the Sidekick class, the UI. In TypeScript: tools, class, Hono API and React page.
  • More tools are a list entry away: search, Wikipedia, files in a sandbox, push, Python.
  • Async setup lives in setup(), not the constructor; cleanup() must close the browser.
  • Prompt fixes matter: the date as context, the print() hint, "benefit of the doubt" for the evaluator.
  • One Sidekick per session (gr.State with ui.load, or a server-side map) keeps users apart.
